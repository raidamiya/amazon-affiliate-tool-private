#!/usr/bin/env python3 # -*- coding: utf-8 -*- """ Twitter Analytics Tool using Twikit 自分のツイートを分析してJSON形式で保存します """ import asyncio import json import os from datetime import datetime, timedelta from collections import Counter import re from twikit import Client # ===== 設定 ===== TWITTER_USERNAME = "raida_bt" # 例: "example_user" TWITTER_EMAIL = "raida.bt@gmail.com" # 例: "email@example.com" TWITTER_PASSWORD = "raida5963" # 例: "password123" MAX_TWEETS = 100 # 取得するツイート数（最大） COOKIES_FILE = "twitter_cookies.json" OUTPUT_FILE = "tweet_analytics.json" # ===== Twitterクライアント ===== client = Client("ja-JP") async def login(): """Twitterにログイン""" print("🔐 Twitterにログイン中...") # クッキーが存在する場合は再利用 if os.path.exists(COOKIES_FILE): try: client.load_cookies(COOKIES_FILE) await client.user_id() # 認証確認 print("✅ 保存済みクッキーでログインしました") return True except Exception as e: print(f"⚠️ クッキーが無効です。再ログインします...") # 新規ログイン try: await client.login( auth_info_1=TWITTER_USERNAME, auth_info_2=TWITTER_EMAIL, password=TWITTER_PASSWORD ) client.save_cookies(COOKIES_FILE) print("✅ ログインに成功しました") return True except Exception as e: print(f"❌ ログイン失敗: {e}") return False async def get_my_tweets(max_tweets=100): """自分のツイートを取得""" print(f"\n📥 最新{max_tweets}件のツイートを取得中...") try: # 自分のユーザーIDを取得 user = await client.user() user_id = user.id print(f"ユーザー: @{user.screen_name}") # 自分のツイートを取得 tweets = await client.get_user_tweets(user_id, 'Tweets', count=max_tweets) tweet_list = [] count = 0 for tweet in tweets: count += 1 # ハッシュタグを抽出 hashtags = re.findall(r'#(\w+)', tweet.text) # 時間帯を計算 created_at = datetime.strptime(tweet.created_at, '%a %b %d %H:%M:%S %z %Y') hour = created_at.hour # ツイートデータをまとめる tweet_data = { 'id': tweet.id, 'text': tweet.text, 'created_at': tweet.created_at, 'created_at_datetime': created_at.isoformat(), 'hour': hour, 'view_count': tweet.view_count or 0, 'retweet_count': tweet.retweet_count or 0, 'favorite_count': tweet.favorite_count or 0, 'reply_count': tweet.reply_count or 0, 'bookmark_count': tweet.bookmark_count or 0, 'hashtags': hashtags, 'has_media': bool(tweet.media), 'is_retweet': hasattr(tweet, 'retweeted_tweet') and tweet.retweeted_tweet is not None } tweet_list.append(tweet_data) print(f"[{count}/{max_tweets}] {created_at.strftime('%Y-%m-%d %H:%M')} | インプ: {tweet.view_count:,} | いいね: {tweet.favorite_count:,}") if count >= max_tweets: break print(f"\n✅ {len(tweet_list)}件のツイートを取得しました") return tweet_list except Exception as e: print(f"❌ ツイート取得エラー: {e}") return [] def analyze_tweets(tweets): """ツイートを分析""" print("\n📊 ツイートを分析中...") if not tweets: print("❌ 分析するツイートがありません") return None # 基本統計 total_tweets = len(tweets) total_impressions = sum(t['view_count'] for t in tweets) total_likes = sum(t['favorite_count'] for t in tweets) total_retweets = sum(t['retweet_count'] for t in tweets) total_replies = sum(t['reply_count'] for t in tweets) total_bookmarks = sum(t['bookmark_count'] for t in tweets) avg_impressions = total_impressions / total_tweets if total_tweets > 0 else 0 avg_likes = total_likes / total_tweets if total_tweets > 0 else 0 avg_retweets = total_retweets / total_tweets if total_tweets > 0 else 0 # エンゲージメント率 engagement_rate = (total_likes + total_retweets + total_replies) / total_impressions * 100 if total_impressions > 0 else 0 # 時間帯別分析 hourly_stats = {} for hour in range(24): hour_tweets = [t for t in tweets if t['hour'] == hour] if hour_tweets: hourly_stats[hour] = { 'count': len(hour_tweets), 'avg_impressions': sum(t['view_count'] for t in hour_tweets) / len(hour_tweets), 'avg_likes': sum(t['favorite_count'] for t in hour_tweets) / len(hour_tweets), 'avg_retweets': sum(t['retweet_count'] for t in hour_tweets) / len(hour_tweets) } # ハッシュタグ分析 all_hashtags = [] for tweet in tweets: all_hashtags.extend(tweet['hashtags']) hashtag_counter = Counter(all_hashtags) top_hashtags = hashtag_counter.most_common(20) # ハッシュタグ別パフォーマンス hashtag_performance = {} for tag, count in top_hashtags: tagged_tweets = [t for t in tweets if tag in t['hashtags']] if tagged_tweets: hashtag_performance[tag] = { 'count': count, 'avg_impressions': sum(t['view_count'] for t in tagged_tweets) / len(tagged_tweets), 'avg_likes': sum(t['favorite_count'] for t in tagged_tweets) / len(tagged_tweets), 'avg_engagement_rate': ( sum(t['favorite_count'] + t['retweet_count'] + t['reply_count'] for t in tagged_tweets) / sum(t['view_count'] for t in tagged_tweets) * 100 ) if sum(t['view_count'] for t in tagged_tweets) > 0 else 0 } # トップパフォーマンスツイート top_tweets_by_impressions = sorted(tweets, key=lambda x: x['view_count'], reverse=True)[:10] top_tweets_by_engagement = sorted( tweets, key=lambda x: (x['favorite_count'] + x['retweet_count'] + x['reply_count']) / max(x['view_count'], 1), reverse=True )[:10] # 分析結果をまとめる analysis = { 'summary': { 'total_tweets': total_tweets, 'total_impressions': total_impressions, 'total_likes': total_likes, 'total_retweets': total_retweets, 'total_replies': total_replies, 'total_bookmarks': total_bookmarks, 'avg_impressions': round(avg_impressions, 2), 'avg_likes': round(avg_likes, 2), 'avg_retweets': round(avg_retweets, 2), 'engagement_rate': round(engagement_rate, 2) }, 'hourly_stats': hourly_stats, 'hashtag_performance': hashtag_performance, 'top_tweets_by_impressions': [ { 'text': t['text'][:100] + '...' if len(t['text']) > 100 else t['text'], 'impressions': t['view_count'], 'likes': t['favorite_count'], 'retweets': t['retweet_count'], 'created_at': t['created_at_datetime'] } for t in top_tweets_by_impressions ], 'top_tweets_by_engagement': [ { 'text': t['text'][:100] + '...' if len(t['text']) > 100 else t['text'], 'impressions': t['view_count'], 'likes': t['favorite_count'], 'retweets': t['retweet_count'], 'engagement_rate': round((t['favorite_count'] + t['retweet_count'] + t['reply_count']) / max(t['view_count'], 1) * 100, 2), 'created_at': t['created_at_datetime'] } for t in top_tweets_by_engagement ], 'all_tweets': tweets } # 結果を表示 print("\n" + "="*60) print("📊 分析結果") print("="*60) print(f"総ツイート数: {total_tweets:,}") print(f"総インプレッション: {total_impressions:,}") print(f"総いいね: {total_likes:,}") print(f"総リツイート: {total_retweets:,}") print(f"総リプライ: {total_replies:,}") print(f"総ブックマーク: {total_bookmarks:,}") print(f"平均インプレッション: {avg_impressions:,.2f}") print(f"平均いいね: {avg_likes:,.2f}") print(f"エンゲージメント率: {engagement_rate:.2f}%") print(f"\n🏆 トップ5ハッシュタグ:") for tag, count in top_hashtags[:5]: perf = hashtag_performance.get(tag, {}) print(f" #{tag}: {count}回 | 平均インプ: {perf.get('avg_impressions', 0):,.0f}") return analysis async def main(): """メイン処理""" print("🚀 Twitter分析ツール") print("="*60) # ログイン if not await login(): return # ツイート取得 tweets = await get_my_tweets(MAX_TWEETS) if not tweets: return # 分析 analysis = analyze_tweets(tweets) if not analysis: return # 結果を保存 print(f"\n💾 結果を {OUTPUT_FILE} に保存中...") with open(OUTPUT_FILE, 'w', encoding='utf-8') as f: json.dump(analysis, f, ensure_ascii=False, indent=2) print(f"✅ 保存完了！") print(f"\n📊 ダッシュボードで確認するには:") print(f" dashboard.html をブラウザで開いてください") if __name__ == "__main__": asyncio.run(main())
